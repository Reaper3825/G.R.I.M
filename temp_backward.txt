0012: // CUDA kernel for Flash Attention backward pass
0013: // Recomputes attention scores during backward to avoid storing them
0014: template<int BLOCK_SIZE_Q, int BLOCK_SIZE_KV, int HEAD_DIM>
0015: __global__ void flashAttentionBackwardKernel(
0016:     const float* __restrict__ Q,           // [batch, heads, seq_len, head_dim]
0017:     const float* __restrict__ K,           // [batch, heads, seq_len, head_dim]
0018:     const float* __restrict__ V,           // [batch, heads, seq_len, head_dim]
0019:     const float* __restrict__ output,      // [batch, heads, seq_len, head_dim]
0020:     const float* __restrict__ grad_output, // [batch, heads, seq_len, head_dim]
0021:     float* __restrict__ grad_Q,            // [batch, heads, seq_len, head_dim]
0022:     float* __restrict__ grad_K,            // [batch, heads, seq_len, head_dim]
0023:     float* __restrict__ grad_V,            // [batch, heads, seq_len, head_dim]
0024:     const float* __restrict__ alibi_slopes, // [heads] or nullptr
0025:     int batch_size,
0026:     int num_heads,
0027:     int seq_len,
0028:     int head_dim,
0029:     bool use_alibi,
0030:     float alibi_scale
0031: ) {
0032:     // Shared memory for blocks
0033:     __shared__ float Q_smem[BLOCK_SIZE_Q][HEAD_DIM];
0034:     __shared__ float K_smem[BLOCK_SIZE_KV][HEAD_DIM];
0035:     __shared__ float V_smem[BLOCK_SIZE_KV][HEAD_DIM];
0036:     __shared__ float dO_smem[BLOCK_SIZE_Q][HEAD_DIM];
0037:     
0038:     // Shared memory for attention scores and gradients
0039:     __shared__ float S_smem[BLOCK_SIZE_Q][BLOCK_SIZE_KV];
0040:     __shared__ float P_smem[BLOCK_SIZE_Q][BLOCK_SIZE_KV];
0041:     __shared__ float dP_smem[BLOCK_SIZE_Q][BLOCK_SIZE_KV];
0042:     
0043:     const int batch_idx = blockIdx.z;
0044:     const int head_idx = blockIdx.y;
0045:     const int q_block_idx = blockIdx.x;
0046:     
0047:     const int tid = threadIdx.x;
0048:     const int num_threads = blockDim.x;
0049:     
0050:     // Global offsets
0051:     const int qkv_offset = batch_idx * num_heads * seq_len * head_dim + 
0052:                            head_idx * seq_len * head_dim;
0053:     
0054:     const float* Q_head = Q + qkv_offset;
0055:     const float* K_head = K + qkv_offset;
0056:     const float* V_head = V + qkv_offset;
0057:     const float* O_head = output + qkv_offset;
0058:     const float* dO_head = grad_output + qkv_offset;
0059:     float* dQ_head = grad_Q + qkv_offset;
0060:     float* dK_head = grad_K + qkv_offset;
0061:     float* dV_head = grad_V + qkv_offset;
0062:     
0063:     const int q_start = q_block_idx * BLOCK_SIZE_Q;
0064:     const int q_end = min(q_start + BLOCK_SIZE_Q, seq_len);
0065:     const int q_size = q_end - q_start;
0066:     
0067:     float alibi_slope = 0.0f;
0068:     if (use_alibi && alibi_slopes != nullptr) {
0069:         alibi_slope = alibi_slopes[head_idx] * alibi_scale;
0070:     }
0071:     
0072:     const float scale = 1.0f / sqrtf(static_cast<float>(head_dim));
0073:     
0074:     // Load Q and dO blocks
0075:     for (int i = tid; i < q_size * head_dim; i += num_threads) {
0076:         const int q_row = i / head_dim;
0077:         const int q_col = i % head_dim;
0078:         const int global_idx = (q_start + q_row) * head_dim + q_col;
0079:         Q_smem[q_row][q_col] = Q_head[global_idx];
0080:         dO_smem[q_row][q_col] = dO_head[global_idx];
0081:     }
0082:     __syncthreads();
0083:     
0084:     // Precompute row sums and max values (needed for softmax gradient)
0085:     float row_max[BLOCK_SIZE_Q];
0086:     float row_sum[BLOCK_SIZE_Q];
0087:     
0088:     #pragma unroll
0089:     for (int i = 0; i < BLOCK_SIZE_Q; ++i) {
0090:         row_max[i] = -FLT_MAX;
0091:         row_sum[i] = 0.0f;
0092:     }
0093:     
0094:     // First pass: compute softmax statistics
0095:     const int num_kv_blocks = (seq_len + BLOCK_SIZE_KV - 1) / BLOCK_SIZE_KV;
0096:     
0097:     for (int kv_block_idx = 0; kv_block_idx < num_kv_blocks; ++kv_block_idx) {
0098:         const int kv_start = kv_block_idx * BLOCK_SIZE_KV;
0099:         const int kv_end = min(kv_start + BLOCK_SIZE_KV, seq_len);
0100:         const int kv_size = kv_end - kv_start;
0101:         
0102:         // Load K block
0103:         for (int i = tid; i < kv_size * head_dim; i += num_threads) {
0104:             const int kv_row = i / head_dim;
0105:             const int kv_col = i % head_dim;
0106:             K_smem[kv_row][kv_col] = K_head[(kv_start + kv_row) * head_dim + kv_col];
0107:         }
0108:         __syncthreads();
0109:         
0110:         // Compute attention scores
0111:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0112:             #pragma unroll
0113:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0114:                 float score = 0.0f;
0115:                 
0116:                 #pragma unroll
0117:                 for (int d = 0; d < HEAD_DIM; ++d) {
0118:                     score += Q_smem[q_idx][d] * K_smem[kv_idx][d];
0119:                 }
0120:                 
0121:                 score *= scale;
0122:                 
0123:                 // Apply ALiBi bias
0124:                 if (use_alibi) {
0125:                     const int q_pos = q_start + q_idx;
0126:                     const int kv_pos = kv_start + kv_idx;
0127:                     score += alibi_slope * static_cast<float>(kv_pos - q_pos);
0128:                 }
0129:                 
0130:                 // Apply causal mask
0131:                 const int q_global_pos = q_start + q_idx;
0132:                 const int kv_global_pos = kv_start + kv_idx;
0133:                 if (kv_global_pos > q_global_pos) {
0134:                     score = -FLT_MAX;
0135:                 }
0136:                 
0137:                 S_smem[q_idx][kv_idx] = score;
0138:                 row_max[q_idx] = fmaxf(row_max[q_idx], score);
0139:             }
0140:         }
0141:         __syncthreads();
0142:     }
0143:     
0144:     // Compute row sums for softmax
0145:     for (int kv_block_idx = 0; kv_block_idx < num_kv_blocks; ++kv_block_idx) {
0146:         const int kv_start = kv_block_idx * BLOCK_SIZE_KV;
0147:         const int kv_end = min(kv_start + BLOCK_SIZE_KV, seq_len);
0148:         const int kv_size = kv_end - kv_start;
0149:         
0150:         // Load K block again (recomputation)
0151:         for (int i = tid; i < kv_size * head_dim; i += num_threads) {
0152:             const int kv_row = i / head_dim;
0153:             const int kv_col = i % head_dim;
0154:             K_smem[kv_row][kv_col] = K_head[(kv_start + kv_row) * head_dim + kv_col];
0155:         }
0156:         __syncthreads();
0157:         
0158:         // Recompute scores and accumulate sum
0159:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0160:             #pragma unroll
0161:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0162:                 float score = 0.0f;
0163:                 
0164:                 #pragma unroll
0165:                 for (int d = 0; d < HEAD_DIM; ++d) {
0166:                     score += Q_smem[q_idx][d] * K_smem[kv_idx][d];
0167:                 }
0168:                 
0169:                 score *= scale;
0170:                 
0171:                 if (use_alibi) {
0172:                     const int q_pos = q_start + q_idx;
0173:                     const int kv_pos = kv_start + kv_idx;
0174:                     score += alibi_slope * static_cast<float>(kv_pos - q_pos);
0175:                 }
0176:                 
0177:                 const int q_global_pos = q_start + q_idx;
0178:                 const int kv_global_pos = kv_start + kv_idx;
0179:                 if (kv_global_pos > q_global_pos) {
0180:                     score = -FLT_MAX;
0181:                 }
0182:                 
0183:                 const float prob = expf(score - row_max[q_idx]);
0184:                 row_sum[q_idx] += prob;
0185:             }
0186:         }
0187:         __syncthreads();
0188:     }
0189:     
0190:     // Second pass: compute gradients
0191:     for (int kv_block_idx = 0; kv_block_idx < num_kv_blocks; ++kv_block_idx) {
0192:         const int kv_start = kv_block_idx * BLOCK_SIZE_KV;
0193:         const int kv_end = min(kv_start + BLOCK_SIZE_KV, seq_len);
0194:         const int kv_size = kv_end - kv_start;
0195:         
0196:         // Load K and V blocks
0197:         for (int i = tid; i < kv_size * head_dim; i += num_threads) {
0198:             const int kv_row = i / head_dim;
0199:             const int kv_col = i % head_dim;
0200:             const int global_idx = (kv_start + kv_row) * head_dim + kv_col;
0201:             K_smem[kv_row][kv_col] = K_head[global_idx];
0202:             V_smem[kv_row][kv_col] = V_head[global_idx];
0203:         }
0204:         __syncthreads();
0205:         
0206:         // Recompute attention probabilities
0207:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0208:             #pragma unroll
0209:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0210:                 float score = 0.0f;
0211:                 
0212:                 #pragma unroll
0213:                 for (int d = 0; d < HEAD_DIM; ++d) {
0214:                     score += Q_smem[q_idx][d] * K_smem[kv_idx][d];
0215:                 }
0216:                 
0217:                 score *= scale;
0218:                 
0219:                 if (use_alibi) {
0220:                     const int q_pos = q_start + q_idx;
0221:                     const int kv_pos = kv_start + kv_idx;
0222:                     score += alibi_slope * static_cast<float>(kv_pos - q_pos);
0223:                 }
0224:                 
0225:                 const int q_global_pos = q_start + q_idx;
0226:                 const int kv_global_pos = kv_start + kv_idx;
0227:                 if (kv_global_pos > q_global_pos) {
0228:                     score = -FLT_MAX;
0229:                 }
0230:                 
0231:                 const float prob = expf(score - row_max[q_idx]) / row_sum[q_idx];
0232:                 P_smem[q_idx][kv_idx] = prob;
0233:             }
0234:         }
0235:         __syncthreads();
0236:         
0237:         // Compute dP = dO @ V^T
0238:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0239:             #pragma unroll
0240:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0241:                 float dp = 0.0f;
0242:                 
0243:                 #pragma unroll
0244:                 for (int d = 0; d < HEAD_DIM; ++d) {
0245:                     dp += dO_smem[q_idx][d] * V_smem[kv_idx][d];
0246:                 }
0247:                 
0248:                 dP_smem[q_idx][kv_idx] = dp;
0249:             }
0250:         }
0251:         __syncthreads();
0252:         
0253:         // Debug: Check dP
0254:         if (tid == 0 && q_block_idx == 0 && batch_idx == 0 && head_idx == 0 && kv_block_idx == 0) {
0255:             printf("[DEBUG] dP_smem[0][0]=%f, dO_smem[0][0]=%f, V_smem[0][0]=%f, P_smem[0][0]=%f\n", 
0256:                    dP_smem[0][0], dO_smem[0][0], V_smem[0][0], P_smem[0][0]);
0257:         }
0258:         
0259:         // Compute dS using softmax backward: dS = P * (dP - sum(dP * P))
0260:         float dp_sum[BLOCK_SIZE_Q];
0261:         #pragma unroll
0262:         for (int i = 0; i < BLOCK_SIZE_Q; ++i) {
0263:             dp_sum[i] = 0.0f;
0264:         }
0265:         
0266:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0267:             #pragma unroll
0268:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0269:                 dp_sum[q_idx] += dP_smem[q_idx][kv_idx] * P_smem[q_idx][kv_idx];
0270:             }
0271:         }
0272:         
0273:         for (int q_idx = tid; q_idx < q_size; q_idx += num_threads) {
0274:             #pragma unroll
0275:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0276:                 const float ds = P_smem[q_idx][kv_idx] * (dP_smem[q_idx][kv_idx] - dp_sum[q_idx]);
0277:                 S_smem[q_idx][kv_idx] = ds * scale;  // Include scaling factor
0278:             }
0279:         }
0280:         __syncthreads();
0281:         
0282:         // Debug: Print first value of S_smem to verify it's non-zero
0283:         if (tid == 0 && q_block_idx == 0 && batch_idx == 0 && head_idx == 0 && kv_block_idx == 0) {
0284:             printf("[Flash Attn Backward DEBUG] S_smem[0][0]=%f, q_size=%d, kv_size=%d, scale=%f\n", 
0285:                    S_smem[0][0], q_size, kv_size, scale);
0286:         }
0287:         
0288:         // Accumulate dQ = dS @ K (write directly per-thread, no atomics needed within block)
0289:         for (int i = tid; i < q_size * head_dim; i += num_threads) {
0290:             const int q_idx = i / head_dim;
0291:             const int d = i % head_dim;
0292:             
0293:             float dq = 0.0f;
0294:             
0295:             for (int kv_idx = 0; kv_idx < kv_size; ++kv_idx) {
0296:                 dq += S_smem[q_idx][kv_idx] * K_smem[kv_idx][d];
0297:             }
0298:             
0299:             // Use atomicAdd because multiple KV blocks contribute to same Q gradient
0300:             atomicAdd(&dQ_head[(q_start + q_idx) * head_dim + d], dq);
0301:         }
0302:         
0303:         // Compute and accumulate dK = dS^T @ Q (using atomics)
0304:         for (int i = tid; i < kv_size * head_dim; i += num_threads) {
0305:             const int kv_idx = i / head_dim;
0306:             const int d = i % head_dim;
0307:             
0308:             float dk = 0.0f;
0309:             for (int q_idx = 0; q_idx < q_size; ++q_idx) {
0310:                 dk += S_smem[q_idx][kv_idx] * Q_smem[q_idx][d];
0311:             }
0312:             
0313:             atomicAdd(&dK_head[(kv_start + kv_idx) * head_dim + d], dk);
0314:         }
0315:         
0316:         // Compute and accumulate dV = P^T @ dO (using atomics)
0317:         for (int i = tid; i < kv_size * head_dim; i += num_threads) {
0318:             const int kv_idx = i / head_dim;
0319:             const int d = i % head_dim;
0320:             
0321:             float dv = 0.0f;
0322:             #pragma unroll
0323:             for (int q_idx = 0; q_idx < q_size; ++q_idx) {
0324:                 dv += P_smem[q_idx][kv_idx] * dO_smem[q_idx][d];
0325:             }
0326:             
0327:             atomicAdd(&dV_head[(kv_start + kv_idx) * head_dim + d], dv);
0328:         }
0329:         
0330:         __syncthreads();
0331:     }
0332:     // dQ is written directly via atomicAdd in the loop above, no final write needed
0333: }

0335: // Host function for Flash Attention backward pass
0336: void flashAttentionBackward(
0337:     const float* Q,
0338:     const float* K,
0339:     const float* V,
0340:     const float* output,
0341:     const float* grad_output,
0342:     float* grad_Q,
0343:     float* grad_K,
0344:     float* grad_V,
0345:     const float* alibi_slopes,
0346:     const FlashAttentionConfig& config
0347: ) {
0348:     // Zero out gradient buffers (required for atomic additions)
0349:     const size_t total_elements = config.batch_size * config.num_heads * 
0350:                                   config.seq_len * config.head_dim;
0351:     
0352:     cudaMemset(grad_Q, 0, total_elements * sizeof(float));
0353:     cudaMemset(grad_K, 0, total_elements * sizeof(float));
0354:     cudaMemset(grad_V, 0, total_elements * sizeof(float));
0355:     
0356:     int block_q = config.block_size_q;
0357:     int block_kv = config.block_size_kv;
0358:     
0359:     if (block_q <= 0 || block_kv <= 0) {
0360:         computeOptimalBlockSizes(config.seq_len, config.head_dim, block_q, block_kv);
0361:     }
0362:     
0363:     const int num_q_blocks = (config.seq_len + block_q - 1) / block_q;
0364:     
0365:     dim3 grid(num_q_blocks, config.num_heads, config.batch_size);
0366:     const int num_threads = 256;
0367:     
0368:     cudaStream_t stream = config.stream ? config.stream : 0;
0369:     
0370:     // Template dispatch - optimized 48x48 blocks
0371:     if (config.head_dim == 32 && block_q == 48 && block_kv == 48) {
0372:         flashAttentionBackwardKernel<48, 48, 32><<<grid, num_threads, 0, stream>>>(
0373:             Q, K, V, output, grad_output, grad_Q, grad_K, grad_V, alibi_slopes,
0374:             config.batch_size, config.num_heads, config.seq_len, config.head_dim,
0375:             config.use_alibi, config.alibi_scale
0376:         );
0377:     } else if (config.head_dim == 64 && block_q == 48 && block_kv == 48) {
0378:         flashAttentionBackwardKernel<48, 48, 64><<<grid, num_threads, 0, stream>>>(
0379:             Q, K, V, output, grad_output, grad_Q, grad_K, grad_V, alibi_slopes,
0380:             config.batch_size, config.num_heads, config.seq_len, config.head_dim,
0381:             config.use_alibi, config.alibi_scale
0382:         );
0383:     } else {
0384:         // Fallback to 32 head_dim
0385:         flashAttentionBackwardKernel<48, 48, 32><<<grid, num_threads, 0, stream>>>(
0386:             Q, K, V, output, grad_output, grad_Q, grad_K, grad_V, alibi_slopes,
0387:             config.batch_size, config.num_heads, config.seq_len, config.head_dim,
0388:             config.use_alibi, config.alibi_scale
0389:         );
0390:     }
0391:     
0392:     cudaError_t err = cudaGetLastError();
0393:     if (err != cudaSuccess) {
0394:         printf("Flash Attention Backward Error: %s\n", cudaGetErrorString(err));
0395:     }
0396: }

0399:     const float* Q,
0400:     const float* K,
0401:     const float* V,
0402:     const float* output,
0403:     const float* grad_output,
0404:     float* grad_Q,

0406:     float* grad_V,
0407:     const float* alibi_slopes,
0408:     void* workspace,
0409:     const FlashAttentionConfig& config
0410: ) {

0413:                           alibi_slopes, config);
0414: }
