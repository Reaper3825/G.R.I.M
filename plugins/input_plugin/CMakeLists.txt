cmake_minimum_required(VERSION 3.22)
project(input_plugin LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../plugin_build.cmake")
find_package(nlohmann_json CONFIG REQUIRED HINTS "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows/share/nlohmann_json")
add_library(input_plugin SHARED input_plugin.cpp input_service.cpp InputController.cpp)
grim_configure_plugin(input_plugin)
if(WIN32)
 target_link_libraries(input_plugin PRIVATE user32)
endif()
target_link_libraries(input_plugin PRIVATE nlohmann_json::nlohmann_json)

option(GRIM_INPUT_PROVIDER_TESTS "Build device-free input tests" OFF)
if(GRIM_INPUT_PROVIDER_TESTS)
 enable_testing()
 add_executable(input_provider_tests "${GRIM_PLUGIN_ROOT}/tests/input_provider_tests.cpp" input_plugin.cpp input_service.cpp)
 target_compile_features(input_provider_tests PRIVATE cxx_std_17)
 target_include_directories(input_provider_tests PRIVATE "${GRIM_PLUGIN_ROOT}" "${GRIM_PLUGIN_ROOT}/core")
 target_compile_definitions(input_provider_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(input_provider_tests PRIVATE nlohmann_json::nlohmann_json)
 add_test(NAME input_provider COMMAND input_provider_tests)
endif()
