cmake_minimum_required(VERSION 3.22)
project(console_plugin LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../plugin_build.cmake")
find_package(nlohmann_json CONFIG REQUIRED HINTS "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows/share/nlohmann_json")
add_library(console_plugin SHARED console_plugin.cpp console_panel.cpp)
grim_configure_plugin(console_plugin
    TOOL_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/console_plugin.tools.json")
target_compile_features(console_plugin PRIVATE cxx_std_20)
target_include_directories(console_plugin PRIVATE "${GRIM_PLUGIN_ROOT}/ui")
target_link_libraries(console_plugin PRIVATE nlohmann_json::nlohmann_json)
# Import the one host UI/session implementation; never compile a second copy.
if(TARGET GRIM)
    target_link_libraries(console_plugin PRIVATE GRIM_EXPORT)
elseif(WIN32)
    set(GRIM_HOST_IMPORT_LIBRARY "" CACHE FILEPATH "Matching GRIM host import library for native UI plugins")
    if(NOT EXISTS "${GRIM_HOST_IMPORT_LIBRARY}")
        message(FATAL_ERROR "console_plugin requires GRIM_HOST_IMPORT_LIBRARY from a matching host build")
    endif()
    target_link_libraries(console_plugin PRIVATE "${GRIM_HOST_IMPORT_LIBRARY}")
elseif(APPLE)
    target_link_options(console_plugin PRIVATE "-undefined" "dynamic_lookup")
endif()
