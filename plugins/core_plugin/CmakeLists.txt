cmake_minimum_required(VERSION 3.22)
project(core_plugin LANGUAGES CXX)

# The plugin DLL
add_library(core_plugin SHARED
    ${CMAKE_SOURCE_DIR}/plugins/core_plugin.cpp
)

target_include_directories(core_plugin PRIVATE
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/commands
    ${CMAKE_SOURCE_DIR}/core
    ${CMAKE_SOURCE_DIR}/logger
)

# Build as plugin
target_compile_definitions(core_plugin PRIVATE GRIM_BUILD_PLUGIN)

# IMPORTANT: link the plugin against GRIM import lib so grim_register_command resolves
# (GRIM must be an executable with ENABLE_EXPORTS ON)
target_link_libraries(core_plugin PRIVATE GRIM)

set_target_properties(core_plugin PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/plugins"
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
)
