cmake_minimum_required(VERSION 3.22)
project(application_plugin LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../plugin_build.cmake")
if(WIN32 AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    list(APPEND CMAKE_PREFIX_PATH "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_library(application_plugin SHARED application_plugin.cpp application_tools.cpp
    application_catalog.cpp application_backends.cpp shell_open.cpp)
if(WIN32)
    target_sources(application_plugin PRIVATE application_backend_windows.cpp)
    target_link_libraries(application_plugin PRIVATE shell32 ole32 uuid)
elseif(APPLE)
    enable_language(OBJCXX)
    set_target_properties(application_plugin PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    target_sources(application_plugin PRIVATE application_backend_macos.mm)
    target_link_libraries(application_plugin PRIVATE "-framework AppKit" "-framework Foundation")
else()
    target_sources(application_plugin PRIVATE application_backend_linux.cpp)
endif()
target_link_libraries(application_plugin PRIVATE nlohmann_json::nlohmann_json)
grim_configure_plugin(application_plugin)

# Catalog uses C++20 value comparisons and UTF-8 filesystem paths.
target_compile_features(application_plugin PRIVATE cxx_std_20)
