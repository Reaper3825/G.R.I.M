cmake_minimum_required(VERSION 3.22)
project(application_plugin LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../plugin_build.cmake")
if(WIN32 AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    list(APPEND CMAKE_PREFIX_PATH "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(FlatBuffers CONFIG REQUIRED)
add_library(application_plugin SHARED application_plugin.cpp application_tools.cpp
    application_catalog.cpp application_backends.cpp user_aliases.cpp portable_applications.cpp shell_open.cpp)
if(WIN32)
    target_sources(application_plugin PRIVATE application_backend_windows.cpp)
    target_link_libraries(application_plugin PRIVATE shell32 ole32 uuid)
elseif(APPLE)
    enable_language(OBJCXX)
    set_target_properties(application_plugin PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
    target_sources(application_plugin PRIVATE application_backend_macos.mm)
    target_link_libraries(application_plugin PRIVATE "-framework AppKit" "-framework Foundation")
else()
    target_sources(application_plugin PRIVATE application_backend_linux.cpp)
endif()
target_link_libraries(application_plugin PRIVATE nlohmann_json::nlohmann_json flatbuffers::flatbuffers)
target_compile_definitions(application_plugin PRIVATE GRIM_APPLICATION_ROOT="${GRIM_PLUGIN_ROOT}")
grim_configure_plugin(application_plugin)

# Catalog uses C++20 value comparisons and UTF-8 filesystem paths.
target_compile_features(application_plugin PRIVATE cxx_std_20)

option(GRIM_APPLICATION_ALIAS_TESTS "Build isolated user-alias and portable discovery tests" OFF)
if(GRIM_APPLICATION_ALIAS_TESTS)
 enable_testing()
 add_executable(application_alias_tests "${GRIM_PLUGIN_ROOT}/tests/application_alias_tests.cpp" user_aliases.cpp portable_applications.cpp)
 target_compile_features(application_alias_tests PRIVATE cxx_std_20)
 target_include_directories(application_alias_tests PRIVATE "${GRIM_PLUGIN_ROOT}")
 target_link_libraries(application_alias_tests PRIVATE nlohmann_json::nlohmann_json flatbuffers::flatbuffers)
 add_test(NAME application_aliases COMMAND application_alias_tests)
endif()
