cmake_minimum_required(VERSION 3.22)
project(voice_plugin LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../plugin_build.cmake")
find_package(nlohmann_json CONFIG REQUIRED HINTS "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows/share/nlohmann_json")
find_package(Threads REQUIRED)
find_library(VOICE_WHISPER_LIBRARY NAMES whisper HINTS "${GRIM_PLUGIN_ROOT}/deps/lib" REQUIRED)
find_library(VOICE_PORTAUDIO_LIBRARY NAMES portaudio HINTS "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows/lib" REQUIRED)
file(GLOB VOICE_RUNTIME CONFIGURE_DEPENDS "runtime/*.cpp")
add_library(voice_plugin SHARED voice_plugin.cpp ${VOICE_RUNTIME})
grim_configure_plugin(voice_plugin)
target_compile_features(voice_plugin PRIVATE cxx_std_20)
target_include_directories(voice_plugin PRIVATE "${GRIM_PLUGIN_ROOT}/deps/include" "${GRIM_PLUGIN_ROOT}/vcpkg_installed/x64-windows/include")
target_link_libraries(voice_plugin PRIVATE nlohmann_json::nlohmann_json Threads::Threads ${VOICE_WHISPER_LIBRARY} ${VOICE_PORTAUDIO_LIBRARY})
if(WIN32)
 target_link_libraries(voice_plugin PRIVATE sapi ole32 oleaut32 shlwapi)
endif()
if(APPLE)
 enable_language(OBJCXX)
 target_sources(voice_plugin PRIVATE runtime/voice_tts_apple.mm)
 target_link_libraries(voice_plugin PRIVATE "-framework AVFoundation" "-framework Foundation")
endif()
option(GRIM_VOICE_PORCUPINE "Use an existing local Porcupine SDK" OFF)
if(GRIM_VOICE_PORCUPINE)
 find_library(VOICE_PORCUPINE_LIBRARY NAMES pv_porcupine HINTS "${GRIM_PLUGIN_ROOT}/external/porcupine/lib/windows/amd64" REQUIRED)
 target_include_directories(voice_plugin PRIVATE "${GRIM_PLUGIN_ROOT}/external/porcupine/include")
 target_compile_definitions(voice_plugin PRIVATE GRIM_HAS_PORCUPINE)
 target_link_libraries(voice_plugin PRIVATE ${VOICE_PORCUPINE_LIBRARY})
endif()

option(GRIM_VOICE_PROVIDER_TESTS "Build device-free provider lifecycle tests" OFF)
if(GRIM_VOICE_PROVIDER_TESTS)
 enable_testing()
 add_executable(voice_provider_tests "${GRIM_PLUGIN_ROOT}/tests/voice_provider_tests.cpp")
 target_compile_features(voice_provider_tests PRIVATE cxx_std_17)
 target_compile_definitions(voice_provider_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_include_directories(voice_provider_tests PRIVATE "${GRIM_PLUGIN_ROOT}")
 target_link_libraries(voice_provider_tests PRIVATE nlohmann_json::nlohmann_json ${CMAKE_DL_LIBS})
 add_test(NAME voice_provider COMMAND voice_provider_tests $<TARGET_FILE:voice_plugin>)
endif()
