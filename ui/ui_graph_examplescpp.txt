// ============================================================
// Example 1: Simple Line Graph
// ============================================================

std::shared_ptr<UIGraph> createLineGraph() {
    auto graph = std::make_shared<UIGraph>("Training Loss", GraphType::Line);
    
    // Configure appearance
    graph->setPosition(50, 100);
    graph->setSize(600, 300);
    
    // Set data
    std::vector<DataPoint> data;
    for (int i = 0; i < 100; ++i) {
        float value = 100.0f * std::exp(-i * 0.03f) + (rand() % 10);
        data.push_back(DataPoint(value, "Step " + std::to_string(i)));
    }
    graph->setData(data);
    
    // Customize
    graph->getConfig().showGrid = true;
    graph->getConfig().showValues = false;
    graph->getConfig().animated = true;
    graph->getConfig().primaryColor = 0xFF00AAFF;
    
    return graph;
}

// ============================================================
// Example 4: Multi-Line Graph (Multiple Series)
// ============================================================

std::shared_ptr<UIGraph> createMultiLineGraph() {
    auto graph = std::make_shared<UIGraph>("Training Metrics", GraphType::MultiLine);
    
    graph->setPosition(50, 100);
    graph->setSize(700, 400);
    
    // Add multiple series
    std::vector<DataPoint> trainLoss;
    std::vector<DataPoint> valLoss;
    std::vector<DataPoint> accuracy;
    
    for (int i = 0; i < 50; ++i) {
        trainLoss.push_back(DataPoint(100.0f * std::exp(-i * 0.05f)));
        valLoss.push_back(DataPoint(100.0f * std::exp(-i * 0.04f) + 5));
        accuracy.push_back(DataPoint(100.0f * (1.0f - std::exp(-i * 0.08f))));
    }
    
    graph->addSeries("Train Loss", trainLoss, 0xFF00AAFF);
    graph->addSeries("Val Loss", valLoss, 0xFFFF6600);
    graph->addSeries("Accuracy", accuracy, 0xFF00FF00);
    
    // Configure
    graph->getConfig().showLegend = true;
    graph->getConfig().showGrid = true;
    graph->getConfig().lineThickness = 2.5f;
    
    return graph;
}

// ============================================================
// Example 5: Real-time Updating Graph
// ============================================================

class RealtimeGraphPanel : public UIPanel {
public:
    RealtimeGraphPanel() : UIPanel("Real-time Monitoring", true) {
        graph = std::make_shared<UIGraph>("CPU Usage", GraphType::Area);
        graph->setPosition(20, 50);
        graph->setSize(600, 300);
        
        graph->setAxisRange(0.0f, 100.0f);
        graph->getConfig().maxDataPoints = 100;  // Keep last 100 points
        graph->getConfig().useDownsampling = true;
        graph->getConfig().primaryColor = 0xFF00FFAA;
        
        addChild(graph);
        
        updateTimer = 0.0f;
    }
    
    void update(const InputState& input, float dt) override {
        UIPanel::update(input, dt);
        
        // Update every 0.1 seconds
        updateTimer += dt;
        if (updateTimer >= 0.1f) {
            updateTimer = 0.0f;
            
            // Simulate CPU usage
            float cpuUsage = 30.0f + (rand() % 40);
            graph->addDataPoint(cpuUsage);
        }
    }
    
private:
    std::shared_ptr<UIGraph> graph;
    float updateTimer;
};

// ============================================================
// Example 6: Interactive Graph with Callbacks
// ============================================================

std::shared_ptr<UIGraph> createInteractiveGraph() {
    auto graph = std::make_shared<UIGraph>("Click Points", GraphType::Scatter);
    
    graph->setPosition(50, 100);
    graph->setSize(600, 400);
    
    // Add scatter data
    std::vector<DataPoint> data;
    std::random_device rd;
    std::mt19937 gen(rd());
    std::uniform_real_distribution<> dis(0.0, 100.0);
    
    for (int i = 0; i < 50; ++i) {
        data.push_back(DataPoint(dis(gen), "Point " + std::to_string(i)));
    }
    graph->setData(data);
    
    // Set up hover callback
    graph->setOnPointHover([](int index, const DataPoint& point) {
        LOG_DEBUG("UIGraph", "Hovering over point " + std::to_string(index) + 
                  " with value " + std::to_string(point.value));
    });
    
    // Set up click callback
    graph->setOnPointClick([](int index, const DataPoint& point) {
        LOG_DEBUG("UIGraph", "Clicked point " + std::to_string(index) + 
                  ": " + point.label);
    });
    
    graph->getConfig().pointRadius = 6.0f;
    graph->getConfig().showValues = true;
    
    return graph;
}

    graph->addSeries("Disk I/O", disk, 0xFF00FF00);
    
    graph->getConfig().showLegend = true;
    graph->getConfig().showGrid = true;
    
    return graph;
}

// ============================================================
// Example 10: Performance Optimized Graph
// ============================================================

std::shared_ptr<UIGraph> createOptimizedGraph() {
    auto graph = std::make_shared<UIGraph>("High Frequency Data", GraphType::Line);
    
    graph->setPosition(50, 100);
    graph->setSize(800, 300);
    
    // Simulate high-frequency data (10,000 points)
    std::vector<DataPoint> data;
    data.reserve(10000);
    
    for (int i = 0; i < 10000; ++i) {
        float value = 50.0f + 30.0f * std::sin(i * 0.01f) + (rand() % 10) - 5;
        data.push_back(DataPoint(value));
    }
    graph->setData(data);  // Automatically downsampled to maxDataPoints
    
    // Optimize for performance
    graph->getConfig().maxDataPoints = 500;  // Downsample to 500 points
    graph->getConfig().useDownsampling = true;
    graph->getConfig().showLabels = false;
    graph->getConfig().showValues = false;
    graph->getConfig().animated = false;  // Disable animation for performance
    
    LOG_DEBUG("UIGraph", "Created optimized graph with automatic downsampling");
    
    return graph;
}

// Example Usage in Main Application
// ============================================================

void exampleUsage() {
    // Create a panel to hold graphs
    auto panel = std::make_shared<UIPanel>("Analytics Dashboard", true);
    panel->setPosition(100, 100);
    panel->setSize(900, 700);
    
    // Add a line graph
    auto lineGraph = createLineGraph();
    lineGraph->setPosition(20, 50);
    panel->addChild(lineGraph);
    
    // Add a bar graph below it
    auto barGraph = createBarGraph();
    barGraph->setPosition(20, 380);
    panel->addChild(barGraph);
    
    // Add to UI root
    UIRoot::get().addPanel(panel);
}
